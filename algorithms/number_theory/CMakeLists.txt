foreach(part sieve factorization divisibility modular)
  add_executable(number_theory_${part}_test ${part}_test.cpp)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(number_theory_${part}_test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME number_theory_${part} COMMAND number_theory_${part}_test)
  set_tests_properties(number_theory_${part} PROPERTIES TIMEOUT 60)
  add_executable(number_theory_${part}_example ${part}_example.cpp)
endforeach()
